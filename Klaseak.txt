    public static void ezabatuGaztea(ArrayList<Gaztea> listaBazkideak, Scanner sc) {

        String NAN;
        boolean aurkitua = false;

        System.out.println("Sartu ezabatu nahi duzun NAN zenbakia: ");
        NAN = sc.next();

        for (int i = 0; i < listaBazkideak.size(); i++) {

            if (listaBazkideak.get(i).getNANValor().equals(NAN)) {

                listaBazkideak.remove(i);
                aurkitua = true;
                System.out.println("Gaztea ezabatu da.");
                break;
            }
        }

        if (aurkitua == false) {
            System.out.println("Ez da NAN hori duen gazterik aurkitu.");
