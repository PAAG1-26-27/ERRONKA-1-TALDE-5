            hitzArray[i] = '_';
        }

                if (hitza.charAt(i) == letra) {
                    hitzArray[i] = letra;
                    aurkitua = true;
                }
            }

                if (hitzArray[i] == '_') {
                    irabazi = false;
                }
            }
        }

        if (irabazi) {
            System.out.println("\nZorionak! Irabazi duzu!");
            System.out.println("Hitza: " + hitza);
        } else {
            System.out.println("\nGaldu duzu!");

            System.out.println("Aukera okerra! Sartu 1 edo 2.");
            System.out.print("Aukeratu: ");
            berriro = irakurri.nextInt();

    public void asmatuZenb(){
        //enuntziatua
        System.out.println("\nAusaz 1 eta 100 bitartean dagoen zenbaki bat aukeratuko da eta zuk asmatu behar duzu. 6 saiakera dituzu.\n");
        //aldagaiak
        int asmatuBehar, saiakera, i;
        asmatuBehar = (int) ((Math.random() * 100));
        saiakera = 6;
        //scanner
        Scanner myObj = new Scanner(System.in);
        //while
        while (saiakera > 0) {
            System.out.print("Sartu zenbaki bat: ");
            i = myObj.nextInt();
        //asmatu zenbakia
            if (i == asmatuBehar) {
                saiakera = 0;
                System.out.println("\nZenbakia asmatu duzu, ZORIONAK!!!");
            }
            else if (i > asmatuBehar) {
                System.out.println("\nZenbakia txikiagoa da, saiatu berriro.");
                saiakera = saiakera - 1;
                System.out.println("Saiakera kopurua: " + saiakera + "\n");
            }
            else {
                System.out.println("\nZenbakia handiagoa da, saiatu berriro.");
                saiakera = saiakera - 1;
                System.out.println("Saiakera kopurua: " + saiakera + "\n");
            }
            //erabiltzaileak jolasa galdu du
            if (saiakera == 0 && i != asmatuBehar) {
                System.out.println("Saiakerarik gabe geratu zara, hurrengoan izango da.");
            }//if
        }//while
    }//asmatuZenb

    public static void matematikak() {

    Scanner sc = new Scanner(System.in);

    int aukera;
    int zenb1, zenb2;
    int erantzuna, emaitza;
    int berriro;

    do {

        System.out.println("===== MATEMATIKAK =====");
        System.out.println("1. Gehiketa");
        System.out.println("2. Kenketa");
        System.out.println("3. Biderketa");
        System.out.println("4. Zatiketa");
        System.out.println("Aukeratu joko bat: ");

        aukera = sc.nextInt();

        switch (aukera) {

            case 1:
                zenb1 = (int)(Math.random() * 100) + 1;
                zenb2 = (int)(Math.random() * 100) + 1;

                System.out.println(zenb1 + " + " + zenb2 + " = ?");
                erantzuna = sc.nextInt();

                emaitza = zenb1 + zenb2;

                if (erantzuna == emaitza) {
                    System.out.println("Zuzena!");
                } else {
                    System.out.println("Okerra! Emaitza: " + emaitza);
                }
                break;

            case 2:
                zenb1 = (int)(Math.random() * 100) + 1;
                zenb2 = (int)(Math.random() * 100) + 1;

                if (zenb1 < zenb2) {
                    int temp = zenb1;
                    zenb1 = zenb2;
                    zenb2 = temp;
                }

                System.out.println(zenb1 + " - " + zenb2 + " = ?");
                erantzuna = sc.nextInt();

                emaitza = zenb1 - zenb2;

                if (erantzuna == emaitza) {
                    System.out.println("Zuzena!");
                } else {
                    System.out.println("Okerra! Emaitza: " + emaitza);
                }
                break;

            case 3:
                zenb1 = (int)(Math.random() * 100) + 1;
                zenb2 = (int)(Math.random() * 100) + 1;

                System.out.println(zenb1 + " x " + zenb2 + " = ?");
                erantzuna = sc.nextInt();

                emaitza = zenb1 * zenb2;

                if (erantzuna == emaitza) {
                    System.out.println("Zuzena!");
                } else {
                    System.out.println("Okerra! Emaitza: " + emaitza);
                }
                break;

            case 4:
                zenb2 = (int)(Math.random() * 100) + 1;
                emaitza = (int)(Math.random() * 100) + 1;
                zenb1 = zenb2 * emaitza;

                System.out.println(zenb1 + " / " + zenb2 + " = ?");
                erantzuna = sc.nextInt();

                if (erantzuna == emaitza) {
                    System.out.println("Zuzena!");
                } else {
                    System.out.println("Okerra! Emaitza: " + emaitza);
                }
                break;

            default:
                System.out.println("Aukera okerra!");
                break;
        }

        System.out.println("----------------------");
        System.out.println("Beste partida bat jokatu nahi duzu?");
        System.out.println("1. Bai");
        System.out.println("2. Ez");
        berriro = sc.nextInt();

    } while (berriro == 1);

    System.out.println("Eskerrik asko jolasteagatik!");

}
